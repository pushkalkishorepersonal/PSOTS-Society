// FIX: Create missing identity record
// Run this in the console on the carpooling page

(async () => {
  console.log('🔧 Creating missing identity record...');
  
  const { auth, db } = await import('/js/core/firebase.js');
  const { doc, setDoc, getDoc, collection, query, where, getDocs } = await import('https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js');
  
  const user = auth.currentUser;
  if (!user) {
    alert('Not logged in');
    return;
  }
  
  console.log('User UID:', user.uid);
  console.log('User email:', user.email);
  
  // Get resident document to find flat number
  const q = query(collection(db, 'residents'), where('email', '==', user.email));
  const snap = await getDocs(q);
  
  if (snap.empty) {
    alert('No resident document found!');
    return;
  }
  
  const residentData = snap.docs[0].data();
  const flatNumber = residentData.flatNumber;
  
  console.log('Flat number:', flatNumber);
  
  if (!flatNumber) {
    alert('No flat number in resident document!');
    return;
  }
  
  // Create identity record
  console.log('Creating identities/' + user.uid);
  await setDoc(doc(db, 'identities', user.uid), {
    flatNumber: flatNumber,
    memberId: user.uid,
    createdAt: new Date(),
    createdBy: 'fix_script'
  });
  
  // Create linkedAccounts record (for legacy compatibility)
  console.log('Creating linkedAccounts/' + user.uid);
  await setDoc(doc(db, 'linkedAccounts', user.uid), {
    flatNumber: flatNumber,
    linkedAt: new Date()
  });
  
  console.log('✅ Identity records created!');
  alert('✅ Identity fixed!\n\nFlat: ' + flatNumber + '\n\nReloading page...');
  
  setTimeout(() => location.reload(), 1000);
})();
